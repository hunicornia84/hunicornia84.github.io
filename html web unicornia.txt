  <script>
    const totalSlides = 11;
    const counter = document.getElementById('counter');
    const scrollHint = document.getElementById('scrollHint');

    // Construir puntos
    for (let i = 1; i <= totalSlides; i++) {
      const dot = document.createElement('div');
      dot.className = 'counter-dot' + (i === 1 ? ' active' : '');
      dot.setAttribute('aria-label', 'Ir a sección ' + i);
      dot.addEventListener('click', () => {
        const id = i <= 10 ? 'slide-' + i : 'slide-11';
        document.getElementById(id).scrollIntoView({ behavior: 'smooth' });
      });
      counter.appendChild(dot);
    }

    const dots = counter.querySelectorAll('.counter-dot');

    // Intersection Observer para detectar slide activo
    const slides = document.querySelectorAll('.slide, .slide-final');
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          const idx = Array.from(slides).indexOf(entry.target);
          dots.forEach((d, i) => d.classList.toggle('active', i === idx));
          // Ocultar hint tras el primer slide
          scrollHint.classList.toggle('hidden', idx > 0);
        }
      });
    }, { threshold: 0.6 });

    slides.forEach(s => observer.observe(s));
  </script>
